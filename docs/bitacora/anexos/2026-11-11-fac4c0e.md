fix(seguridad): cerrar el motor de IA de Django y valores por defecto seguros

y sin limite: se podia gastar la cuota de Gemini. Ahora solo los usa Spring:

defecto; sin DJANGO_SECRET_KEY y sin DEBUG no arranca. .env.example lo explica.

+"""Quien puede usar el motor de IA.
+
+El motor de IA solo lo consume Spring Boot. Cada identificacion gasta cuota del
+proveedor (Gemini, OpenAI...), asi que no puede quedar abierto a cualquiera en la red:
+
+- Si `IA_CLAVE_INTERNA` esta configurada, la peticion debe traer el header
+  `X-Ia-Clave` con ese valor (Spring lo envia; ver IaConfig.java).
+- Si no esta configurada, solo se aceptan peticiones desde la propia maquina
+  (127.0.0.1 / ::1), que es como se ejecuta en desarrollo.
+"""
+import hmac
+
+from django.conf import settings
+from rest_framework.permissions import BasePermission
+
+_LOCALES = {"127.0.0.1", "::1"}
+
+
+class SoloClienteInterno(BasePermission):
+    message = "Acceso al motor de IA no autorizado"
+
+    def has_permission(self, request, view):
+        clave = settings.IA_CLAVE_INTERNA
+        if clave:
+            return hmac.compare_digest(clave, request.headers.get("X-Ia-Clave", ""))
+        return request.META.get("REMOTE_ADDR") in _LOCALES

+package com.plantia.config;
+
+import org.springframework.beans.factory.annotation.Value;
+import org.springframework.boot.web.client.RestClientCustomizer;
+import org.springframework.context.annotation.Bean;
+import org.springframework.context.annotation.Configuration;
+
+/**
+ * Clave compartida con el motor de IA de Django (header X-Ia-Clave).
+ * Django solo acepta llamadas con esa clave, o desde la misma maquina si no hay
+ * clave configurada (ver backend-django/ia_engine/permisos.py).
+ * Se aplica al RestClient.Builder que usa IaClient, unico cliente HTTP saliente.
+ */
+@Configuration
+public class IaConfig {
+
+    @Bean
+    RestClientCustomizer claveInternaIa(@Value("${IA_CLAVE_INTERNA:}") String clave) {
+        return builder -> {
+            if (!clave.isBlank()) {
+                builder.defaultHeader("X-Ia-Clave", clave);
+            }
+        };
+    }
+}
